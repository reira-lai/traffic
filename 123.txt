    <script>
        // ================================================================
        // 站碼設定（根據實際資料）
        // ================================================================
        const STOPS = {
            kmb: {
                '59X': 'TM883',   // 屯門碼頭總站
                '59M': 'TM630',   // 蝴蝶邨蝶心樓
                '59A': 'TM630',
                'N260': 'TM883',
                '259B': 'TM630',
                '59S': 'TM883'
            },
            ctb: {
                '962X': '001410',
                '962P': '001410',
                'B3': '001410'
            },
            lwb: {
                'A33': 'TM883'    // 可能不準，僅供參考
            }
        };

        // ================================================================
        // 輔助函數
        // ================================================================
        function getMinutes(etaStr) {
            if (!etaStr) return null;
            const diff = new Date(etaStr) - new Date();
            const mins = Math.round(diff / 60000);
            if (mins <= 0) return '🚀 即將到達';
            if (mins < 1) return '<1 分鐘';
            return mins + ' 分鐘';
        }

        function setEta(elementId, dataArray) {
            const el = document.getElementById(elementId);
            if (!el) return;
            if (!dataArray || dataArray.length === 0) {
                el.innerHTML = '暫無班次';
                el.className = 'eta-badge text-sm text-stone-400';
                return;
            }
            const times = dataArray.slice(0, 3).map(item => getMinutes(item.eta)).filter(Boolean);
            if (times.length === 0) {
                el.innerHTML = '暫無班次';
                el.className = 'eta-badge text-sm text-stone-400';
            } else {
                el.innerHTML = times.join(' · ');
                el.className = 'eta-badge text-sm font-semibold text-stone-800';
            }
        }

        function setError(elementId, errMsg, retryFn) {
            const el = document.getElementById(elementId);
            el.innerHTML = `❌ ${errMsg}`;
            el.className = 'eta-badge text-sm text-red-500 cursor-pointer';
            el.onclick = retryFn;
        }

        // ================================================================
        // API 請求（官方優先，備用第三方）
        // ================================================================

        // ---- 九巴官方 API ----
        async function fetchKMB(route, elementId) {
            const stopId = STOPS.kmb[route];
            if (!stopId) { setError(elementId, '無站碼', () => fetchKMB(route, elementId)); return; }
            try {
                const url = `https://data.etabus.gov.hk/v1/transport/kmb/stop-eta/${stopId}`;
                const resp = await fetch(url);
                if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
                const json = await resp.json();
                const filtered = (json.data || []).filter(item => item.route === route);
                setEta(elementId, filtered);
            } catch (err) {
                console.warn(`KMB ${route} 官方失敗，嘗試第三方...`, err);
                // 備用：data.hkbus.app
                try {
                    const url2 = `https://data.hkbus.app/eta/kmb/${route}/${stopId}`;
                    const resp2 = await fetch(url2);
                    if (!resp2.ok) throw new Error(`HTTP ${resp2.status}`);
                    const data2 = await resp2.json();
                    setEta(elementId, data2);
                } catch (err2) {
                    setError(elementId, 'API 錯誤', () => fetchKMB(route, elementId));
                }
            }
        }

        // ---- 城巴官方 API ----
        async function fetchCTB(route, elementId) {
            const stopId = STOPS.ctb[route];
            if (!stopId) { setError(elementId, '無站碼', () => fetchCTB(route, elementId)); return; }
            try {
                const url = `https://rt.data.gov.hk/v1/transport/citybus/eta/ctb/${stopId}/${route}`;
                const resp = await fetch(url);
                if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
                const json = await resp.json();
                setEta(elementId, json.data || []);
            } catch (err) {
                console.warn(`CTB ${route} 官方失敗，嘗試第三方...`, err);
                try {
                    const url2 = `https://data.hkbus.app/eta/ctb/${route}/${stopId}`;
                    const resp2 = await fetch(url2);
                    if (!resp2.ok) throw new Error(`HTTP ${resp2.status}`);
                    const data2 = await resp2.json();
                    setEta(elementId, data2);
                } catch (err2) {
                    setError(elementId, 'API 錯誤', () => fetchCTB(route, elementId));
                }
            }
        }

        // ---- 龍運（官方無通用 ETA，先用第三方） ----
        async function fetchLWB(route, elementId) {
            const stopId = STOPS.lwb[route];
            if (!stopId) { setError(elementId, '無站碼', () => fetchLWB(route, elementId)); return; }
            try {
                const url = `https://data.hkbus.app/eta/lwb/${route}/${stopId}`;
                const resp = await fetch(url);
                if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
                const data = await resp.json();
                setEta(elementId, data);
            } catch (err) {
                setError(elementId, 'API 錯誤', () => fetchLWB(route, elementId));
            }
        }

        // ---- 港鐵巴士（官方） ----
        async function fetchMTR(route, elementId) {
            try {
                const url = `https://rt.data.gov.hk/v1/transport/mtr/bus/eta?route=${route}`;
                const resp = await fetch(url);
                if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
                const json = await resp.json();
                let etas = [];
                if (json.bus && json.bus.eta) {
                    const all = json.bus.eta;
                    const filtered = all.filter(item => 
                        item.stopName && (item.stopName.includes('碼頭') || item.stopName.includes('Ferry'))
                    );
                    etas = filtered.length > 0 ? filtered : all;
                }
                setEta(elementId, etas);
            } catch (err) {
                setError(elementId, 'API 錯誤', () => fetchMTR(route, elementId));
            }
        }

        // ---- 輕鐵（官方） ----
        async function fetchLR(elementId, stationId = '120') {
            try {
                const url = `https://rt.data.gov.hk/v1/transport/mtr/lrt/getSchedule?station_id=${stationId}`;
                const resp = await fetch(url);
                if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
                const json = await resp.json();
                const etas = [];
                if (json.platform && json.platform.length) {
                    json.platform.forEach(p => {
                        if (p.routeList) {
                            p.routeList.forEach(r => {
                                if (r.trains) {
                                    r.trains.forEach(t => {
                                        if (t.time) etas.push({ eta: t.time });
                                    });
                                }
                            });
                        }
                    });
                }
                etas.sort((a,b) => new Date(a.eta) - new Date(b.eta));
                setEta(elementId, etas);
            } catch (err) {
                setError(elementId, 'API 錯誤', () => fetchLR(elementId, stationId));
            }
        }

        // ================================================================
        // 總控 + 自動更新
        // ================================================================
        let countdownValue = 30;
        let countdownTimer = null;

        async function fetchAllData() {
            const icon = document.getElementById('refresh-icon');
            icon.className = 'refresh-spin';
            const now = new Date();
            document.getElementById('update-time').textContent = now.toLocaleTimeString('zh-HK', { hour12: false });

            await Promise.all([
                fetchKMB('59X', 'eta-59x'),
                fetchKMB('59M', 'eta-59m'),
                fetchKMB('59A', 'eta-59a'),
                fetchKMB('N260', 'eta-n260'),
                fetchKMB('259B', 'eta-259b'),
                fetchKMB('59S', 'eta-59s'),
                fetchLWB('A33', 'eta-a33'),
                fetchCTB('962X', 'eta-962x'),
                fetchCTB('962P', 'eta-962p'),
                fetchCTB('B3', 'eta-b3'),
                fetchMTR('K52', 'eta-k52'),
                fetchMTR('506', 'eta-506'),
                fetchLR('eta-lr', '120')
            ]);

            icon.className = '';
            countdownValue = 30;
            document.getElementById('countdown').textContent = countdownValue;
        }

        function startCountdown() {
            if (countdownTimer) clearInterval(countdownTimer);
            countdownTimer = setInterval(() => {
                countdownValue--;
                if (countdownValue <= 0) {
                    fetchAllData();
                    countdownValue = 30;
                }
                document.getElementById('countdown').textContent = countdownValue;
            }, 1000);
        }

        window.onload = function() {
            fetchAllData();
            startCountdown();
        };
    </script>
